cmake_minimum_required(VERSION 4.4.3)
project(competitive)

file(GLOB_RECURSE SOLUTIONS CONFIGURE_DEPENDS "*.cpp")
foreach(input ${SOLUTIONS})
  if (${input} MATCHES "/build/")
   continue()
  endif()
  
  get_filename_component(exedir "${input}" DIRECTORY)
  get_filename_component(exe "${input}" NAME_WE)
  string(REPLACE "${CMAKE_SOURCE_DIR}/" "" exedir "${exedir}")
  string(REPLACE "/" "_" exedir "${exedir}")
  add_executable("${exedir}_${exe}" ${input})
endforeach()
